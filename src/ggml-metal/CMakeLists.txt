find_library(FOUNDATION_LIBRARY Foundation REQUIRED)
find_library(METAL_FRAMEWORK    Metal      REQUIRED)
find_library(METALKIT_FRAMEWORK MetalKit   REQUIRED)

message(STATUS "Metal framework found")

ggml_add_backend_library(ggml-metal
                         ggml-metal.cpp
                         ggml-metal-device.m
                         ggml-metal-device.cpp
                         ggml-metal-common.cpp
                         ggml-metal-context.m
                         ggml-metal-ops.cpp
                        )

target_link_libraries(ggml-metal PRIVATE
                      ${FOUNDATION_LIBRARY}
                      ${METAL_FRAMEWORK}
                      ${METALKIT_FRAMEWORK}
                      )

if (GGML_METAL_NDEBUG)
    add_compile_definitions(GGML_METAL_NDEBUG)
endif()

set(METALLIB_COMMON "${CMAKE_CURRENT_SOURCE_DIR}/../ggml-common.h")
if (GGML_METAL_EMBED_LIBRARY)
    enable_language(ASM)

    add_compile_definitions(GGML_METAL_EMBED_LIBRARY)

    set(METALLIB_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/ggml-metal.metal")
    set(METALLIB_IMPL   "${CMAKE_CURRENT_SOURCE_DIR}/ggml-metal-impl.h")

    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/autogenerated")

    # Merge ggml-common.h and ggml-metal.metal into one translation unit. Prefer
    # embedding precompiled libraries: compiling this large source at process
    # startup leaves hundreds of MiB in Apple's compiler heap even for tiny
    # graphs. Xcode 26 ships the Metal compiler as an optional toolchain, while
    # older Xcodes expose it through the default toolchain, so probe both.
    set(METALLIB_EMBED_ASM        "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-embed.s")
    set(METALLIB_SOURCE_EMBED     "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-embed.metal")
    set(METALLIB_SOURCE_EMBED_TMP "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-embed.metal.tmp")

    set(METAL_DEFAULT_RESULT 1)
    set(METALLIB_TOOLCHAIN_RESULT 1)
    execute_process(
        COMMAND xcrun --toolchain Metal -f metal
        RESULT_VARIABLE METAL_TOOLCHAIN_RESULT
        OUTPUT_QUIET ERROR_QUIET)
    if (METAL_TOOLCHAIN_RESULT EQUAL 0)
        set(METAL_XCRUN_TOOLCHAIN_ARGS --toolchain Metal)
    else()
        execute_process(
            COMMAND xcrun -f metal
            RESULT_VARIABLE METAL_DEFAULT_RESULT
            OUTPUT_QUIET ERROR_QUIET)
        if (METAL_DEFAULT_RESULT EQUAL 0)
            set(METAL_XCRUN_TOOLCHAIN_ARGS)
        endif()
    endif()

    if (METAL_TOOLCHAIN_RESULT EQUAL 0 OR METAL_DEFAULT_RESULT EQUAL 0)
        execute_process(
            COMMAND xcrun ${METAL_XCRUN_TOOLCHAIN_ARGS} -f metallib
            RESULT_VARIABLE METALLIB_TOOLCHAIN_RESULT
            OUTPUT_QUIET ERROR_QUIET)
    endif()

    # Finding `metal` is insufficient: Xcode versions before Metal 4 accept
    # the tool lookup but reject the tensor variant later in the build. Probe
    # the exact language/type surface up front and retain source embedding when
    # the installed toolchain cannot compile every runtime-selected variant.
    set(METAL_BINARY_EMBED_SUPPORTED FALSE)
    if ((METAL_TOOLCHAIN_RESULT EQUAL 0 OR METAL_DEFAULT_RESULT EQUAL 0) AND
        METALLIB_TOOLCHAIN_RESULT EQUAL 0)
        set(METALLIB_PROBE_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-toolchain-probe.metal")
        set(METALLIB_BF16_PROBE_AIR "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-toolchain-bf16.air")
        set(METALLIB_TENSOR_PROBE_AIR "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-toolchain-tensor.air")
        file(WRITE "${METALLIB_PROBE_SOURCE}" [=[
#include <metal_stdlib>
#if defined(GGML_METAL_PROBE_TENSOR)
#include <metal_tensor>
#include <MetalPerformancePrimitives/MetalPerformancePrimitives.h>
#endif
using namespace metal;
#if defined(GGML_METAL_PROBE_BF16)
kernel void ggml_metal_toolchain_probe(device bfloat * value [[buffer(0)]]) { value[0] = value[0]; }
#else
kernel void ggml_metal_toolchain_probe(device float * value [[buffer(0)]]) { value[0] = value[0]; }
#endif
]=])
        execute_process(
            COMMAND xcrun ${METAL_XCRUN_TOOLCHAIN_ARGS} -sdk macosx metal
                    -std=metal3.1 -DGGML_METAL_PROBE_BF16=1
                    -c "${METALLIB_PROBE_SOURCE}" -o "${METALLIB_BF16_PROBE_AIR}"
            RESULT_VARIABLE METALLIB_BF16_PROBE_RESULT
            OUTPUT_QUIET ERROR_QUIET)
        execute_process(
            COMMAND xcrun ${METAL_XCRUN_TOOLCHAIN_ARGS} -sdk macosx metal
                    -std=metal4.0 -DGGML_METAL_PROBE_BF16=1 -DGGML_METAL_PROBE_TENSOR=1
                    -c "${METALLIB_PROBE_SOURCE}" -o "${METALLIB_TENSOR_PROBE_AIR}"
            RESULT_VARIABLE METALLIB_TENSOR_PROBE_RESULT
            OUTPUT_QUIET ERROR_QUIET)
        if (METALLIB_BF16_PROBE_RESULT EQUAL 0 AND METALLIB_TENSOR_PROBE_RESULT EQUAL 0)
            set(METAL_BINARY_EMBED_SUPPORTED TRUE)
        endif()
    endif()

    if (METAL_BINARY_EMBED_SUPPORTED)
        add_compile_definitions(GGML_METAL_EMBED_LIBRARY_BINARY)

        set(METALLIB_BASELINE_AIR "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-baseline.air")
        set(METALLIB_BASELINE_LIB "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-baseline.metallib")
        set(METALLIB_BF16_AIR     "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-bf16.air")
        set(METALLIB_BF16_LIB     "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-bf16.metallib")
        set(METALLIB_TENSOR_AIR   "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-tensor.air")
        set(METALLIB_TENSOR_LIB   "${CMAKE_CURRENT_BINARY_DIR}/autogenerated/ggml-metal-tensor.metallib")

        if (GGML_METAL_SHADER_DEBUG)
            set(METALLIB_XC_FLAGS -fno-fast-math -fno-inline)
        else()
            set(METALLIB_XC_FLAGS -O3)
        endif()
        if (GGML_METAL_MACOSX_VERSION_MIN)
            list(APPEND METALLIB_XC_FLAGS -mmacosx-version-min=${GGML_METAL_MACOSX_VERSION_MIN})
        endif()

        add_custom_command(
            OUTPUT "${METALLIB_EMBED_ASM}"
            COMMAND echo "Precompiling embedded Metal libraries"
            COMMAND sed -e "/__embed_ggml-common.h__/r ${METALLIB_COMMON}"       -e "/__embed_ggml-common.h__/d"         < "${METALLIB_SOURCE}"           > "${METALLIB_SOURCE_EMBED_TMP}"
            COMMAND sed -e "/\#include \"ggml-metal-impl.h\"/r ${METALLIB_IMPL}" -e "/\#include \"ggml-metal-impl.h\"/d" < "${METALLIB_SOURCE_EMBED_TMP}" > "${METALLIB_SOURCE_EMBED}"
            COMMAND xcrun ${METAL_XCRUN_TOOLCHAIN_ARGS} -sdk macosx metal ${METALLIB_XC_FLAGS} -std=metal3.0 -DGGML_METAL_EMBED_LIBRARY=1 -c "${METALLIB_SOURCE_EMBED}" -o "${METALLIB_BASELINE_AIR}"
            COMMAND xcrun ${METAL_XCRUN_TOOLCHAIN_ARGS} -sdk macosx metallib "${METALLIB_BASELINE_AIR}" -o "${METALLIB_BASELINE_LIB}"
            COMMAND xcrun ${METAL_XCRUN_TOOLCHAIN_ARGS} -sdk macosx metal ${METALLIB_XC_FLAGS} -std=metal3.1 -DGGML_METAL_EMBED_LIBRARY=1 -DGGML_METAL_HAS_BF16=1 -c "${METALLIB_SOURCE_EMBED}" -o "${METALLIB_BF16_AIR}"
            COMMAND xcrun ${METAL_XCRUN_TOOLCHAIN_ARGS} -sdk macosx metallib "${METALLIB_BF16_AIR}" -o "${METALLIB_BF16_LIB}"
            COMMAND xcrun ${METAL_XCRUN_TOOLCHAIN_ARGS} -sdk macosx metal ${METALLIB_XC_FLAGS} -std=metal4.0 -DGGML_METAL_EMBED_LIBRARY=1 -DGGML_METAL_HAS_BF16=1 -DGGML_METAL_HAS_TENSOR=1 -c "${METALLIB_SOURCE_EMBED}" -o "${METALLIB_TENSOR_AIR}"
            COMMAND xcrun ${METAL_XCRUN_TOOLCHAIN_ARGS} -sdk macosx metallib "${METALLIB_TENSOR_AIR}" -o "${METALLIB_TENSOR_LIB}"
            COMMAND echo ".section __DATA,__ggml_metallib"                    >  "${METALLIB_EMBED_ASM}"
            COMMAND echo ".globl _ggml_metallib_baseline_start"              >> "${METALLIB_EMBED_ASM}"
            COMMAND echo "_ggml_metallib_baseline_start:"                    >> "${METALLIB_EMBED_ASM}"
            COMMAND echo .incbin "\"${METALLIB_BASELINE_LIB}\""              >> "${METALLIB_EMBED_ASM}"
            COMMAND echo ".globl _ggml_metallib_baseline_end"                >> "${METALLIB_EMBED_ASM}"
            COMMAND echo "_ggml_metallib_baseline_end:"                      >> "${METALLIB_EMBED_ASM}"
            COMMAND echo ".globl _ggml_metallib_bf16_start"                  >> "${METALLIB_EMBED_ASM}"
            COMMAND echo "_ggml_metallib_bf16_start:"                        >> "${METALLIB_EMBED_ASM}"
            COMMAND echo .incbin "\"${METALLIB_BF16_LIB}\""                 >> "${METALLIB_EMBED_ASM}"
            COMMAND echo ".globl _ggml_metallib_bf16_end"                    >> "${METALLIB_EMBED_ASM}"
            COMMAND echo "_ggml_metallib_bf16_end:"                          >> "${METALLIB_EMBED_ASM}"
            COMMAND echo ".globl _ggml_metallib_tensor_start"                >> "${METALLIB_EMBED_ASM}"
            COMMAND echo "_ggml_metallib_tensor_start:"                      >> "${METALLIB_EMBED_ASM}"
            COMMAND echo .incbin "\"${METALLIB_TENSOR_LIB}\""               >> "${METALLIB_EMBED_ASM}"
            COMMAND echo ".globl _ggml_metallib_tensor_end"                  >> "${METALLIB_EMBED_ASM}"
            COMMAND echo "_ggml_metallib_tensor_end:"                        >> "${METALLIB_EMBED_ASM}"
            DEPENDS ../ggml-common.h ggml-metal.metal ggml-metal-impl.h
            COMMENT "Generate embedded precompiled Metal libraries"
            VERBATIM)
    else()
        message(STATUS "A complete Metal 3.1/4.0 toolchain is unavailable; embedding source for runtime compilation")
        add_custom_command(
            OUTPUT "${METALLIB_EMBED_ASM}"
            COMMAND echo "Embedding Metal source library"
            COMMAND sed -e "/__embed_ggml-common.h__/r ${METALLIB_COMMON}"       -e "/__embed_ggml-common.h__/d"         < "${METALLIB_SOURCE}"           > "${METALLIB_SOURCE_EMBED_TMP}"
            COMMAND sed -e "/\#include \"ggml-metal-impl.h\"/r ${METALLIB_IMPL}" -e "/\#include \"ggml-metal-impl.h\"/d" < "${METALLIB_SOURCE_EMBED_TMP}" > "${METALLIB_SOURCE_EMBED}"
            COMMAND echo ".section __DATA,__ggml_metallib"          >  "${METALLIB_EMBED_ASM}"
            COMMAND echo ".globl _ggml_metallib_start"              >> "${METALLIB_EMBED_ASM}"
            COMMAND echo "_ggml_metallib_start:"                    >> "${METALLIB_EMBED_ASM}"
            COMMAND echo .incbin "\"${METALLIB_SOURCE_EMBED}\""     >> "${METALLIB_EMBED_ASM}"
            COMMAND echo ".globl _ggml_metallib_end"                >> "${METALLIB_EMBED_ASM}"
            COMMAND echo "_ggml_metallib_end:"                      >> "${METALLIB_EMBED_ASM}"
            DEPENDS ../ggml-common.h ggml-metal.metal ggml-metal-impl.h
            COMMENT "Generate assembly for embedded Metal source"
            VERBATIM)
    endif()

    target_sources(ggml-metal PRIVATE "${METALLIB_EMBED_ASM}")
else()
    # copy metal files to bin directory
    configure_file(../ggml-common.h  ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/ggml-common.h     COPYONLY)
    configure_file(ggml-metal.metal  ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/ggml-metal.metal  COPYONLY)
    configure_file(ggml-metal-impl.h ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/ggml-metal-impl.h COPYONLY)

    if (GGML_METAL_SHADER_DEBUG)
        # custom command to do the following:
        #   xcrun -sdk macosx metal    -fno-fast-math -c ggml-metal.metal -o ggml-metal.air
        #   xcrun -sdk macosx metallib                   ggml-metal.air   -o default.metallib
        #
        # note: this is the only way I found to disable fast-math in Metal. it's ugly, but at least it works
        #       disabling fast math is needed in order to pass tests/test-backend-ops
        # note: adding -fno-inline fixes the tests when using MTL_SHADER_VALIDATION=1
        # note: unfortunately, we have to call it default.metallib instead of ggml.metallib
        #       ref: https://github.com/ggml-org/whisper.cpp/issues/1720
        # note: adding -g causes segmentation fault during compile
        #set(XC_FLAGS -fno-fast-math -fno-inline -g)
        set(XC_FLAGS -fno-fast-math -fno-inline)
    else()
        set(XC_FLAGS -O3)
    endif()

    # Append macOS metal versioning flags
    if (GGML_METAL_MACOSX_VERSION_MIN)
        message(STATUS "Adding  -mmacosx-version-min=${GGML_METAL_MACOSX_VERSION_MIN} flag to metal compilation")
        list   (APPEND XC_FLAGS -mmacosx-version-min=${GGML_METAL_MACOSX_VERSION_MIN})
    endif()

    if (GGML_METAL_STD)
        message(STATUS "Adding  -std=${GGML_METAL_STD} flag to metal compilation")
        list   (APPEND XC_FLAGS -std=${GGML_METAL_STD})
    endif()

    add_custom_command(
        OUTPUT ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/default.metallib
        COMMAND xcrun -sdk macosx metal ${XC_FLAGS} -c ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/ggml-metal.metal -o - |
                xcrun -sdk macosx metallib        - -o ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/default.metallib
        COMMAND rm -f ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/ggml-common.h
        COMMAND rm -f ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/ggml-metal.metal
        DEPENDS ggml-metal.metal ${METALLIB_COMMON}
        COMMENT "Compiling Metal kernels"
        )

    # FIXME: only add to the ggml-metal target?
    add_custom_target(
        ggml-metal-lib ALL
        DEPENDS ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/default.metallib
        )
endif() # GGML_METAL_EMBED_LIBRARY

if (NOT GGML_METAL_EMBED_LIBRARY)
    install(
        FILES src/ggml-metal/ggml-metal.metal
        PERMISSIONS
            OWNER_READ
            OWNER_WRITE
            GROUP_READ
            WORLD_READ
        DESTINATION ${CMAKE_INSTALL_BINDIR})

        install(
            FILES ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/default.metallib
            DESTINATION ${CMAKE_INSTALL_BINDIR}
        )
endif()
